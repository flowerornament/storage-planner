   ```rust
   };
   ```

9. Add `--item-id` to the bail message listing available flags.

10. Resolve item_id before the transaction (same pattern as add):
    ```rust
    let resolved_item_id = if let Some(iid) = item_id {
        let item = resolve_catalog_item(db, iid)?;
        Some(item.id)
    } else {
        None
    };
    ```

11. Add to the after-state builder:
    ```rust
    if let Some(ref iid) = resolved_item_id {
        after.item_id = Some(iid.clone());
    ```

    ```rust

        )?;
    ```
